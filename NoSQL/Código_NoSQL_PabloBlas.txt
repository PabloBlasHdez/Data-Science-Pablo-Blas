db.movies.aggregate([
  // Obtenemos el año más reciente (maxYear)
  {
    $group: {
      _id: null,
      maxYear: { $max: "$year" }
    }
  },
  // Calculamos el límite inferior (últimos 20 años >> restamos 19)
  {
    $project: {
      _id: 0,
      maxYear: 1,
      minYear: { $subtract: ["$maxYear", 19] }
    }
  },
  // Contamos las películas dentro del rango [minYear, maxYear]
  {
    $lookup: {
      from: "movies",
      let: { maxY: "$maxYear", minY: "$minYear" },
      pipeline: [
        {
          $match: {
            $expr: {
              $and: [
                { $gte: ["$year", "$$minY"] },
                { $lte: ["$year", "$$maxY"] }
              ]
            }
          }
        },
        { $count: "totalPeliculasUltimos20" }
      ],
      as: "conteo"
    }
  },
  // Resultado final
  {
    $project: {
      maxYear: 1,
      minYear: 1,
      totalPeliculasUltimos20: {
        $arrayElemAt: ["$conteo.totalPeliculasUltimos20", 0]
      }
    }
  }
]);

db.movies.aggregate([
  // Obtenemos el año más reciente (maxYear)
  {
    $group: {
      _id: null,
      maxYear: { $max: "$year" }
    }
  },
  // Calculamos el límite inferior (últimos 20 años >> restamos 19)
  {
    $project: {
      _id: 0,
      maxYear: 1,
      minYear: { $subtract: ["$maxYear", 19] }
    }
  },
  // Unimos (join) la colección de películas consigo misma (self-join) para poder aplicar un filtro ($match) que use las variables calculadas.
  {
    $lookup: {
      from: "movies",                                             // La misma colección
      pipeline: [
        { $match: { _id: { $exists: true } } }          // Devuelve todos los documentos de 'movies'
      ],
      as: "peliculas"
    }
  },
  // Desenrollamos el array de películas, creando un documento por cada película
  {
    $unwind: "$peliculas"
  },
  // Filtramos las películas que cumplen con el rango de [minYear, maxYear]
  {
    $match: {
      $expr: {
        $and: [
          { $gte: ["$peliculas.year", "$minYear"] }, // year >= minYear
          { $lte: ["$peliculas.year", "$maxYear"] }  // year <= maxYear
        ]
      }
    }
  },
  // Proyectamos el documento final de la película, eliminando las variables de rango
  {
    $replaceRoot: {
      newRoot: "$peliculas"
    }
  }
]);

  {
    $group: {
      _id: "$year",
      totalPeliculas: { $sum: 1 }
    }
  },
  // Encontramos el máximo número de películas en un año
  {
    $group: {
      _id: null,
      maxPeliculas: { $max: "$totalPeliculas" },
      data: { $push: { year: "$_id", totalPeliculas: "$totalPeliculas" } }
    }
  },
  // Filtramos solo los años que tienen el máximo número de películas
  {
    $unwind: "$data"
  },
  {
    $match: {
      $expr: { $eq: ["$data.totalPeliculas", "$maxPeliculas"] }
    }
  },
  // Proyectamos el resultado final limpio
  {
    $project: {
      _id: 0,
      year: "$data.year",
      totalPeliculas: "$data.totalPeliculas"
    }
  }
]);

Query libre 2 ---------------------------------------------------
db.movies.aggregate([
  // Expandimos el array 'cast' para generar un documento por cada aparición de un actor
  { $unwind: "$cast" },
  // Filtramos los documentos para excluir actores con el valor "Undefined"
  { $match: { cast: { $ne: "Undefined" } } },
  // Agrupamos por la combinación (Década + Actor) y cuenta la frecuencia
  // La década se calcula con operadores matemáticos: año - (año % 10)
  { $group: {
      _id: {
        decada: { $subtract: ["$year", { $mod: ["$year", 10] }] }, 
        actor: "$cast"
      },
      count_movies: { $sum: 1 }
    } },
  // Ordenamos los resultados por década y, luego, por conteo descendente (el actor más popular de cada década se posiciona primero)
  { $sort: { "_id.decada": 1, "count_movies": -1 } },
  // Volvemos a agrupar solo por la década. Usamos $first para capturar el actor y el conteo del primer documento de cada grupo
  { $group: {
      _id: "$_id.decada",
      actor_top: { $first: "$_id.actor" },
      max_movies: { $first: "$count_movies" }
    } },
  // Formateamos la salida para transformar la década numérica en un rango de texto (YYYY-YYYY)
  { $project: {
      _id: 0,
      Decada: {
        $concat: [
          { $toString: "$_id" },
          "-",
          { $toString: { $add: ["$_id", 9] } }
        ]
      },
      Actor_top: "$actor_top",
      Num_Peliculas: "$max_movies",
    } },
  // Ordenamos el resultado final por la década cronológicamente
  { $sort: { "Decada": 1 } }
]);


Query libre 3 ---------------------------------------------------
db.movies.aggregate([
  // Eliminamos películas con arrays de géneros vacíos o “Undefined”
  { $match: { "genres": { $nin: ["Undefined", "", null], $exists: true, $ne: [] } } },
  // Creamos una copia de ‘genres’ para posteriormente realizar un doble unwind.
  { $addFields: { genres_copy: "$genres" } },
  // Descomponemos y generamos todas las combinaciones posibles de géneros dentro de cada película.
  { $unwind: "$genres" },
  { $unwind: "$genres_copy" },
  // Normalizamos usando $match con $expr para asegurar que:
  // >> El Género A sea menor que el Género B ($lt) para tratar (A, B) y (B, A) como un único par.
  // >> Se excluyan los pares donde el género es el mismo (A, A).
  { $match: { $expr: { $lt: ["$genres", "$genres_copy"] } } },
  // Concatenamos el par de géneros en un único campo (ej: "Action | Drama").
  { $project: {
    _id: 0,
    genre_pair: { $concat: ["$genres", " | ", "$genres_copy"] }
  } },
  // Agrupamos por pares y contamos la frecuencia de co-ocurrencia.
  { $group: {
    _id: "$genre_pair",
    co_occurrence: { $sum: 1 }
  } },
  // Ordenamos para encontrar los pares más frecuentes.
  { $sort: { co_occurrence: -1 } },
  // Mostramos solo los 5 pares principales.
  { $limit: 5 }
]);
